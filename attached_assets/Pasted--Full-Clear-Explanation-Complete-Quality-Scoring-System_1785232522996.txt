3. Final Quality Scoring Algorithm (Clean & Dynamic)

function calculateQualityScore(data: any) {
  const {
    mcGrowth = 0,
    volumeIntensity = 0,
    holderVelocity = 0,
    kolSmartScore = 0,        // 0-100
    liquidityHealth = 0,
    top10Pct = 1.0,
    marketCapUsd = 0,
    holderCount = 0
  } = data;

  // Base Score
  let score = 
    (mcGrowth * 0.35) +
    (volumeIntensity * 0.25) +
    (holderVelocity * 0.20) +
    (kolSmartScore * 0.12) +
    (liquidityHealth * 0.08);

  // === Risk Penalties ===
  if (top10Pct > 0.78) score -= 28;
  else if (top10Pct > 0.68) score -= 18;
  else if (top10Pct > 0.58) score -= 8;

  if (marketCapUsd  75) score += 9;
  if (kolSmartScore > 85) score += 6;

  if (volumeIntensity > 90 && mcGrowth > 85) score += 7;

  score = Math.max(5, Math.min(100, Math.round(score)));

  // Label
  let label = "Weak";
  if (score >= 82) label = "Elite";
  else if (score >= 72) label = "Excellent";
  else if (score >= 62) label = "Strong";
  else if (score >= 52) label = "Good";
  else if (score >= 40) label = "Average";
  else if (score >= 25) label = "Speculative";

  return {
    score,
    label,
    breakdown: {
      mcGrowth: Math.round(mcGrowth * 0.35),
      volume: Math.round(volumeIntensity * 0.25),
      holderVelocity: Math.round(holderVelocity * 0.20),
      kolSmart: Math.round(kolSmartScore * 0.12),
      liquidity: Math.round(liquidityHealth * 0.08),
      finalScore: score
    }
  };
}
